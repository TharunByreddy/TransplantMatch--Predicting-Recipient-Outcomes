data$outcome <- ifelse(!is.na(data$composite_dt) &

🧼 Remove NZV + High Correlation

data_numeric <- data %>%
  select(where(is.numeric), where(is.factor)) %>%
  filter(!is.na(outcome))

nzv <- nearZeroVar(data_numeric)
data_filtered <- if (length(nzv) > 0) data_numeric[, -nzv] else data_numeric

cor_matrix <- cor(data_filtered %>% select(where(is.numeric)), use = "pairwise.complete.obs")
high_cor <- findCorrelation(cor_matrix, cutoff = 0.95)
if (length(high_cor) > 0) {
  data_filtered <- data_filtered[, -high_cor]
}

✅ Safe Variable Screening for MICE (One-by-One Test)

safe_vars <- c()
data_filtered <- data_filtered %>% mutate(outcome = as.integer(outcome))
other_vars <- setdiff(names(data_filtered), "outcome")

for (v in other_vars) {
  test_data <- data_filtered[, c("outcome", v), drop = FALSE]
  result <- tryCatch({
    mice(test_data, maxit = 1, printFlag = FALSE)
    TRUE
  }, error = function(e) FALSE)
  
  if (result) {
    safe_vars <- c(safe_vars, v)
  }
}

data_safe <- data_filtered[, c("outcome", safe_vars)]

📉 Lasso Regression

x_train <- model.matrix(outcome ~ ., train)[, -1]
y_train <- train$outcome

cv_lasso <- cv.glmnet(x_train, y_train, alpha = 1, family = "binomial")
lasso_model <- glmnet(x_train, y_train, alpha = 1, lambda = cv_lasso$lambda.min)

x_test <- model.matrix(outcome ~ ., test)[, -1]
lasso_pred <- predict(lasso_model, newx = x_test, type = "response")

lasso_roc <- prediction(as.numeric(lasso_pred), test$outcome)
lasso_auc <- performance(lasso_roc, "auc")@y.values[[1]]

🌲 Random Forest

rf_model <- randomForest(as.factor(outcome) ~ ., data = train, ntree = 100)
rf_pred <- predict(rf_model, test, type = "prob")[, 2]
rf_roc <- prediction(rf_pred, test$outcome)
rf_auc <- performance(rf_roc, "auc")@y.values[[1]]
